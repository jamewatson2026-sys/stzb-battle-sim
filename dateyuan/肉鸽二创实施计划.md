### Task 1: 关卡状态机

**Files:**
- Create: `src/rogue/types.ts`, `src/rogue/run.ts`
- Test: `tests/rogue/run.test.ts`

**Interfaces:**
- Consumes: 无（本任务是地基）
- Produces:
  - `type Outcome = 'win' | 'lose' | 'draw'`
  - `interface RunState { schemaVersion: 1; seed: number; level: number; levelCount: number; lives: number; status: 'playing' | 'cleared' | 'dead' }`
  - `createRun(seed: number, levelCount?: number, lives?: number): RunState`
  - `applyOutcome(state: RunState, outcome: Outcome): RunState`

**规则（来自 spec §3）**：`win` → 关卡 +1，超过 `levelCount` 则 `status='cleared'`；`lose`（大营被斩首）→ 命数 -1，关号不变，命数归零则 `status='dead'`；`draw`（真平）→ 命数与关号都不变。终局后再结算一律返回原状态（幂等）。

```ts
  });

  it('不可变：不改原对象', () => {
```

```ts
// src/rogue/types.ts
export type Outcome = 'win' | 'lose' | 'draw';

export interface RunState {
  schemaVersion: 1;
  seed: number;
  /** 当前关卡号，1 起 */
  level: number;
  levelCount: number;
  lives: number;
  status: 'playing' | 'cleared' | 'dead';
}
```

```ts
// src/rogue/run.ts
import type { Outcome, RunState } from './types';

export function createRun(seed: number, levelCount = 3, lives = 3): RunState {
  return { schemaVersion: 1, seed, level: 1, levelCount, lives, status: 'playing' };
}

export function applyOutcome(state: RunState, outcome: Outcome): RunState {
  if (state.status !== 'playing') return state;
  if (outcome === 'draw') return state;
  if (outcome === 'lose') {
    const lives = state.lives - 1;
    return { ...state, lives, status: lives <= 0 ? 'dead' : 'playing' };
  }
  const level = state.level + 1;
  return level > state.levelCount
    ? { ...state, level: state.levelCount, status: 'cleared' }
    : { ...state, level };
}
```

```ts
export interface HeroSlot {
  heroId: string;
  maxTroops: number;
  /** 当前剩余兵力（跨关累计；战后回写） */
  troops: number;
}

export interface PoolState {
  heroes: HeroSlot[];
  skills: string[];
  treasures: string[];
  /** 已获得的越界卡 id */
  cards: string[];
}
```

```ts
// src/rogue/pool.ts
import type { HeroSlot, PoolState } from './types';

export function createPool(init: {
  heroes: Array<{ heroId: string; maxTroops: number }>;
  skills?: string[];
}): PoolState {
  return {
    heroes: init.heroes.map((h) => ({ heroId: h.heroId, maxTroops: h.maxTroops, troops: h.maxTroops })),
    skills: [...(init.skills ?? [])],
    treasures: [],
    cards: [],
  };
}

const clamp = (v: number, max: number) => Math.max(0, Math.min(max, Math.round(v)));

export function applyBattleResult(
  pool: PoolState,
  remaining: Array<{ heroId: string; troops: number }>
): PoolState {
  const byId = new Map(remaining.map((r) => [r.heroId, r.troops]));
  return {
    ...pool,
    heroes: pool.heroes.map((h) =>
      byId.has(h.heroId) ? ({ ...h, troops: clamp(byId.get(h.heroId)!, h.maxTroops) } satisfies HeroSlot) : h
    ),
  };
}

export function addCards(pool: PoolState, cardIds: string[]): PoolState {
  const merged = [...new Set([...pool.cards, ...cardIds])];
  return { ...pool, cards: merged };
}
```

```ts
// src/rogue/save.ts
import type { PoolState, RunState } from './types';

export interface StorageLike {
  getItem(key: string): string | null;
  setItem(key: string, value: string): void;
  removeItem(key: string): void;
}

export interface SaveData {
  run: RunState;
  pool: PoolState;
}

export const SAVE_KEY = 'rogue.run.v1';

export function memoryStorage(): StorageLike {
  const m = new Map<string, string>();
  return {
    getItem: (k) => m.get(k) ?? null,
    setItem: (k, v) => void m.set(k, v),
    removeItem: (k) => void m.delete(k),
  };
}

export function saveRun(storage: StorageLike, data: SaveData): void {
  storage.setItem(SAVE_KEY, JSON.stringify(data));
}

const isObj = (v: unknown): v is Record<string, unknown> => typeof v === 'object' && v !== null;

function validRun(v: unknown): v is RunState {
  if (!isObj(v)) return false;
  return (
    v.schemaVersion === 1 &&
    typeof v.seed === 'number' &&
    typeof v.level === 'number' &&
    typeof v.levelCount === 'number' &&
    typeof v.lives === 'number' &&
    (v.status === 'playing' || v.status === 'cleared' || v.status === 'dead')
  );
}

function validPool(v: unknown): v is PoolState {
  if (!isObj(v)) return false;
  if (!Array.isArray(v.heroes) || !Array.isArray(v.skills)) return false;
  if (!Array.isArray(v.treasures) || !Array.isArray(v.cards)) return false;
  return v.heroes.every(
    (h) => isObj(h) && typeof h.heroId === 'string' && typeof h.maxTroops === 'number' && typeof h.troops === 'number'
  );
}

/** 读档：任何异常形状一律返回 null（信任边界，不抛） */
export function loadRun(storage: StorageLike): SaveData | null {
  const raw = storage.getItem(SAVE_KEY);
  if (!raw) return null;
  let parsed: unknown;
  try {
    parsed = JSON.parse(raw);
  } catch {
    return null;
  }
  if (!isObj(parsed)) return null;
  if (!validRun(parsed.run) || !validPool(parsed.pool)) return null;
  return { run: parsed.run, pool: parsed.pool };
}
```

```ts
// src/rogue/variant.ts
import { SKILL_REGISTRY } from '../data/skills';
import type { Skill } from '../engine/types';

export const ROGUE_PREFIX = 'rogue:';

export function isRogueSource(skillId: string): boolean {
  return skillId.startsWith(ROGUE_PREFIX);
}

/** `rogue:<cardId>:<baseSkillId>` → cardId；非越界来源返回 null */
export function rogueCardOf(skillId: string): string | null {
  if (!isRogueSource(skillId)) return null;
  const rest = skillId.slice(ROGUE_PREFIX.length);
  const i = rest.indexOf(':');
  return i > 0 ? rest.slice(0, i) : null;
}

/** 战报可读名（照 engine/treasure-source.ts 的 treasureLabel 模式） */
export function rogueLabel(skillId: string): string {
  const card = rogueCardOf(skillId);
  if (!card) return skillId;
  const base = skillId.slice(ROGUE_PREFIX.length + card.length + 1);
  const baseName = SKILL_REGISTRY[base]?.name ?? base;
  const cardName = card;
  return `【越界·${cardName}】${baseName}`;
}

/** 运行时注册战法变体：拷贝基础战法 → 打补丁 → 以新 id 落表 */
export function registerVariant(cardId: string, baseSkillId: string, patch: Partial<Skill>): string {
  const base = SKILL_REGISTRY[baseSkillId];
  if (!base) throw new Error(`基础战法不存在：${baseSkillId}`);
  const id = `${ROGUE_PREFIX}${cardId}:${baseSkillId}`;
  SKILL_REGISTRY[id] = { ...structuredClone(base), ...patch, id } as Skill;
  return id;
}
```

```ts
// src/rogue/cards.ts
import { SKILL_REGISTRY } from '../data/skills';
import { mainSkillSlot } from '../data/hero-utils';
import type { General, Skill } from '../engine/types';
import { ROGUE_PREFIX, isRogueSource, rogueLabel } from './variant';

export interface RogueCard {
  id: string;
  name: string;
  tier: 'L1' | 'L2' | 'L3';
  desc: string;
}

export const ROGUE_CARDS: RogueCard[] = [
  { id: 'rb_borrow', name: '越俎代庖', tier: 'L1', desc: '可装配任意武将的主战法' },
  { id: 'rb_echo', name: '余音不绝', tier: 'L3', desc: '我军每发动一次主动战法，立即全额再发动一次' },
];

/** rb_borrow：把 from 的主战法塞进 to 的对应槽位（to 自己的主战法保留） */
export function borrowMainSkill(to: General, from: General): General {
  const src = from.mainSkillId;
  if (!src) return to;
  const slot = mainSkillSlot(src);
  if (!slot) return to;
  if (to[slot].includes(src)) return to;
  return { ...to, [slot]: [...to[slot], src] } as General;
}

const ECHO_ID = `${ROGUE_PREFIX}rb_echo:echo`;

/** rb_echo：合成「全额再发动」指挥战法；幂等 */
export function ensureEchoSkill(): string | null {
  if (SKILL_REGISTRY[ECHO_ID]) return ECHO_ID;
  // 拿一个现成 command 战法当骨架 → 清空它自己的输出 → 只留 allyRecast 行为
  const skeleton = Object.values(SKILL_REGISTRY).find((s) => s.type === 'command' && !isRogueSource(s.id));
  if (!skeleton) return null;
  SKILL_REGISTRY[ECHO_ID] = {
    ...structuredClone(skeleton),
    id: ECHO_ID,
    name: rogueLabel(ECHO_ID),
    output: [],
    allyRecast: { rate: 100, factor: 1, everyCast: true },
  } as unknown as Skill;
  return ECHO_ID;
}

/** rb_echo：把合成指挥战法挂进该武将的指挥槽（重复挂不叠加） */
export function attachEcho(hero: General): General {
  const id = ensureEchoSkill();
  if (!id || hero.commandSkillIds.includes(id)) return hero;
  return { ...hero, commandSkillIds: [...hero.commandSkillIds, id] };
}
```

```ts
// src/rogue/enemy.ts
import { runBattle } from '../engine/combat';
import type { General } from '../engine/types';
import { judgeOutcome } from '../../web/battleSim';

const sum = (a: number[]) => a.reduce((x, y) => x + y, 0);

/** 我方对敌方的胜率（口径与生产一致：斩首优先，打满则比兵力比） */
export function winRateVs(myTeam: General[], enemyTeam: General[], runs: number, baseSeed: number): number {
  const myCap = sum(myTeam.map((g) => g.maxTroops));
  const enCap = sum(enemyTeam.map((g) => g.maxTroops));
  let wins = 0;
  for (let i = 0; i < runs; i++) {
    const report = runBattle({ myTeam, enemyTeam, seed: baseSeed + i, maxRounds: 8 });
    const judged = judgeOutcome(
      report.result,
      sum(report.finalMyTroops) / myCap,
      sum(report.finalEnemyTroops) / enCap
    );
    if (judged.win) wins++;
  }
  return wins / runs;
}

/** 从候选里挑一支「玩家胜率最接近 target」的敌方阵容 */
export function pickEnemy(
  playerTeam: General[],
  candidates: General[][],
  target: number,
  opts: { runs?: number; baseSeed?: number } = {}
): { team: General[]; winRate: number } {
  if (candidates.length === 0) throw new Error('pickEnemy：候选阵容为空');
  const runs = opts.runs ?? 20;
  const baseSeed = opts.baseSeed ?? 20261007;
  let best = { team: candidates[0], winRate: winRateVs(playerTeam, candidates[0], runs, baseSeed) };
  for (const team of candidates.slice(1)) {
    const wr = winRateVs(playerTeam, team, runs, baseSeed);
    if (Math.abs(wr - target) < Math.abs(best.winRate - target)) best = { team, winRate: wr };
  }
  return best;
}
```

```ts
// src/rogue/rewards.ts
import type { Rng } from '../engine/rng';
import type { PoolState } from './types';
import type { RogueCard } from './cards';

export type RewardKind = 'hero' | 'skill' | 'treasure' | 'card' | 'troops';

export interface RewardOption {
  id: string;
  kind: RewardKind;
  label: string;
  /** 载荷：heroId / skillId / cardId */
  payload: string;
}

export interface RewardCatalogue {
  heroes: Array<{ heroId: string; label: string }>;
  skills: Array<{ skillId: string; label: string }>;
  cards: RogueCard[];
}

/** 越界卡进入奖励池的起始关卡（spec §5.1：越界强度 = f(关卡进度)） */
export const CARD_UNLOCK_LEVEL = 2;

export function rollRewards(
  rng: Rng,
  pool: PoolState,
  level: number,
  catalogue: RewardCatalogue
): RewardOption[] {
  const ownedHeroes = new Set(pool.heroes.map((h) => h.heroId));
  const ownedSkills = new Set(pool.skills);
  const ownedCards = new Set(pool.cards);

  const poolHeroes: RewardOption[] = catalogue.heroes
    .filter((h) => !ownedHeroes.has(h.heroId))
    .map((h) => ({ id: `rw_hero_${h.heroId}`, kind: 'hero', label: `武将·${h.label}`, payload: h.heroId }));
  const poolSkills: RewardOption[] = catalogue.skills
    .filter((s) => !ownedSkills.has(s.skillId))
    .map((s) => ({ id: `rw_skill_${s.skillId}`, kind: 'skill', label: `战法·${s.label}`, payload: s.skillId }));
  const poolCards: RewardOption[] =
    level >= CARD_UNLOCK_LEVEL
      ? catalogue.cards
          .filter((c) => !ownedCards.has(c.id))
          .map((c) => ({ id: `rw_card_${c.id}`, kind: 'card', label: `越界·${c.name}`, payload: c.id }))
      : [];

  const all = [...poolCards, ...poolHeroes, ...poolSkills];
  const picked: RewardOption[] = [];
  const bag = [...all];
  while (picked.length < 3 && bag.length > 0) {
    picked.push(bag.splice(rng.int(bag.length), 1)[0]);
  }
  return picked;
}
```

> `Rng` 的 `int(n)` 返回 `[0, n)` 的整数（见 `src/engine/rng.ts`）。若候选不足 3 个，`rollRewards` 会返回少于 3 个——UI 需容忍（奖励池常见）。

```ts
// src/rogue/loop.ts
import { runBattle } from '../engine/combat';
import type { BattleReport, General, Position } from '../engine/types';
import { judgeOutcome } from '../../web/battleSim';
import { applyOutcome } from './run';
import { applyBattleResult } from './pool';
import { attachEcho, borrowMainSkill } from './cards';
import type { Outcome, PoolState, RunState } from './types';

export interface TeamChoice {
  slots: Array<{
    heroId: string;
    skillIds: string[];
    position: Position;
    /** rb_borrow 的来源武将（借用其主战法） */
    borrowFrom?: string;
  }>;
}

export interface LoopDeps {
  buildGeneral: (heroId: string, skillIds: string[], position: Position) => General;
}

/** 按选择构建我方队伍，并自动施加池子里已拥有的越界卡 */
export function buildMyTeam(pool: PoolState, choice: TeamChoice, deps: LoopDeps): General[] {
  const owned = new Set(pool.cards);
  return choice.slots.map((slot) => {
    let g = deps.buildGeneral(slot.heroId, slot.skillIds, slot.position);
    if (owned.has('rb_echo')) g = attachEcho(g);
    if (owned.has('rb_borrow') && slot.borrowFrom) {
      const from = deps.buildGeneral(slot.borrowFrom, [], '中军');
      g = borrowMainSkill(g, from);
    }
    return g;
  });
}

export interface LevelResult {
  run: RunState;
  pool: PoolState;
  report: BattleReport;
  outcome: Outcome;
}

const sum = (a: number[]) => a.reduce((x, y) => x + y, 0);

/** 结算一关：跑战斗 → 判胜负 → 写回兵力 → 推进状态机 */
export function settleLevel(
  run: RunState,
  pool: PoolState,
  myTeam: General[],
  enemyTeam: General[],
  seedForLevel: number
): LevelResult {
  const report = runBattle({
    myTeam,
    enemyTeam,
    seed: run.seed + seedForLevel * 1000,
    maxRounds: 8,
  });
  const judged = judgeOutcome(
    report.result,
    sum(report.finalMyTroops) / sum(myTeam.map((g) => g.maxTroops)),
    sum(report.finalEnemyTroops) / sum(enemyTeam.map((g) => g.maxTroops))
  );
  const outcome: Outcome = judged.draw ? 'draw' : judged.win ? 'win' : 'lose';
  const nextPool = applyBattleResult(
    pool,
    myTeam.map((g, i) => ({ heroId: pool.heroes[i]?.heroId ?? g.id, troops: report.finalMyTroops[i] ?? 0 }))
  );
  return { run: applyOutcome(run, outcome), pool: nextPool, report, outcome };
}
```

> ⚠️ `settleLevel` 里"我方剩余兵力 → 池子武将"的对应关系依赖 `myTeam` 与 `pool.heroes` 同序。若 `buildMyTeam` 将来改变顺序，必须改成按 `heroId` 显式配对（`repo` 里 `report.finalMyTroops` 顺序 = `myTeam` 顺序，见 `RunRaw` 注释）。**Task 9 的端到端用例会覆盖这条**。

```ts
import { Rng } from '../../src/engine/rng';
import { createRun } from '../../src/rogue/run';
import { createPool, addCards } from '../../src/rogue/pool';
import { loadRun, saveRun, type StorageLike } from '../../src/rogue/save';
import { buildMyTeam, settleLevel, type TeamChoice } from '../../src/rogue/loop';
import { rollRewards, type RewardCatalogue, type RewardOption } from '../../src/rogue/rewards';
import { ROGUE_CARDS } from '../../src/rogue/cards';
import { buildGeneral, SLOTTED_HEROES } from '../heroes';
import { renderIntel } from './intel';
import { renderTeam } from './team';
import { renderReward } from './reward';
import { renderReport } from './report';
import type { PoolState, RunState } from '../../src/rogue/types';
import type { General } from '../../src/engine/types';

const deps = {
  buildGeneral: (heroId: string, skillIds: string[], position: General['position']) =>
    buildGeneral(heroId, skillIds, {}, position, 0, 40),
};

const INITIAL_HEROES = ['h479', 'h29', 'h16'];

export function mountRogue(root: HTMLElement, opts: { storage?: StorageLike } = {}): void {
  const storage = opts.storage ?? window.localStorage;
  const restored = loadRun(storage);
  let run: RunState = restored?.run ?? createRun(Date.now() % 100000, 3, 3);
  let pool: PoolState = restored?.pool ?? createPool({
    heroes: INITIAL_HEROES.map((heroId) => ({ heroId, maxTroops: 9000 })),
    skills: [],
  });
  const rng = new Rng(run.seed);

  const catalogue: RewardCatalogue = {
    heroes: SLOTTED_HEROES.slice(0, 30).map((h) => ({ heroId: h.id, label: h.name })),
    skills: [],
    cards: ROGUE_CARDS,
  };
  const heroList = SLOTTED_HEROES.map((h) => ({ id: h.id, name: h.name }));
  const persist = () => saveRun(storage, { run, pool });
  const enemyFor = (level: number): General[] => {
    const picks = SLOTTED_HEROES.slice(level * 3, level * 3 + 3);
    return picks.map((h, i) =>
      deps.buildGeneral(h.id, [], (['前锋', '中军', '大营'] as const)[i])
    );
  };

  const screenIntel = () => renderIntel(root, { enemy: enemyFor(run.level), level: run.level, lives: run.lives, onFight: screenTeam });
  const screenTeam = () =>
    renderTeam(root, {
      pool,
      heroes: heroList,
      skills: [],
      onConfirm: (choice: TeamChoice) => {
        const myTeam = buildMyTeam(pool, choice, deps);
        const res = settleLevel(run, pool, myTeam, enemyFor(run.level), run.level);
        run = res.run;
        pool = res.pool;
        persist();
        renderReport(root, {
          report: res.report,
          outcome: res.outcome,
          onNext:
            res.outcome === 'win'
              ? () => screenReward()
              : () => (run.status === 'dead' ? screenEnd() : screenIntel()),
        });
      },
    });
  const screenReward = () => {
    const options = rollRewards(rng, pool, run.level, catalogue);
    renderReward(root, {
      options,
      onPick: (o: RewardOption) => {
        if (o.kind === 'card') pool = addCards(pool, [o.payload]);
        if (o.kind === 'hero') pool = { ...pool, heroes: [...pool.heroes, { heroId: o.payload, maxTroops: 9000, troops: 9000 }] };
        if (o.kind === 'skill') pool = { ...pool, skills: [...pool.skills, o.payload] };
        persist();
        run.status === 'cleared' ? screenEnd() : screenIntel();
      },
    });
  };
  const screenEnd = () => {
    root.innerHTML = `<section id="rg-end"><h2>${run.status === 'cleared' ? '通关！' : '全军覆没'}</h2></section>`;
  };

  run.status === 'playing' ? screenIntel() : screenEnd();
  root.setAttribute('data-rogue-ready', '1');
}
```

```ts
// web/rogue/report.ts
import { reportToText } from '../../src/engine/report';
import { computeDetailedStats } from '../../src/engine/stats';
import type { BattleReport } from '../../src/engine/types';
import { isRogueSource, rogueCardOf } from '../../src/rogue/variant';
import { ROGUE_CARDS } from '../../src/rogue/cards';
import type { Outcome } from '../../src/rogue/types';

const cardName = (id: string) => ROGUE_CARDS.find((c) => c.id === id)?.name ?? id;

/** 按 `rogue:` 前缀聚合越界卡贡献（spec §5.4） */
export function rogueContributions(report: BattleReport): Array<{ card: string; casts: number; damage: number }> {
  const acc = new Map<string, { card: string; casts: number; damage: number }>();
  for (const ev of report.events as any[]) {
    const id: string | undefined = ev.skillId ?? ev.sourceId;
    if (!id || !isRogueSource(id)) continue;
    const card = rogueCardOf(id) ?? id;
    const row = acc.get(card) ?? { card, casts: 0, damage: 0 };
    if (ev.type === 'skill_cast') row.casts += 1;
    if (typeof ev.damage === 'number') row.damage += ev.damage;
    acc.set(card, row);
  }
  return [...acc.values()].sort((a, b) => b.damage - a.damage);
}

export function renderReport(
  root: HTMLElement,
  v: { report: BattleReport; outcome: Outcome; onNext: () => void }
): void {
  const rows = rogueContributions(v.report);
  const stats = computeDetailedStats(v.report.events, [] as any);
  const kind = v.outcome === 'win' ? '胜' : v.outcome === 'lose' ? '败' : '平';
  root.innerHTML = `
    <section id="rg-report">
      <h2>战报 · ${kind}</h2>
      <div id="rg-rogue-highlight">
        <h3>越界贡献</h3>
        ${
          rows.length
            ? `<ul>${rows.map((r) => `<li>${cardName(r.card)}：出手 ${r.casts} 次，伤害 ${r.damage}</li>`).join('')}</ul>`
            : '<p>本局未使用越界卡</p>'
        }
```

```ts
// tests/rogue/e2e.test.ts
import { describe, it, expect } from 'vitest';
import { Rng } from '../../src/engine/rng';
import { buildGeneral, SLOTTED_HEROES } from '../../web/heroes';
import { createRun } from '../../src/rogue/run';
import { createPool, addCards } from '../../src/rogue/pool';
import { buildMyTeam, settleLevel } from '../../src/rogue/loop';
import { rollRewards } from '../../src/rogue/rewards';
import { ROGUE_CARDS } from '../../src/rogue/cards';
import { saveRun, loadRun, memoryStorage } from '../../src/rogue/save';
import { rogueContributions } from '../../web/rogue/report';

const deps = { buildGeneral: (id: string, s: string[], p: any) => buildGeneral(id, s, {}, p, 0, 50) };
const POS = ['前锋', '中军', '大营'] as const;

describe('端到端：3 关可玩闭环', () => {
  it('强队能连打 3 关并通关，且中途存档可恢复', () => {
    const storage = memoryStorage();
    const strong = SLOTTED_HEROES.slice(0, 3).map((h) => h.id);
    let run = createRun(20261007, 3, 3);
    let pool = createPool({ heroes: strong.map((heroId) => ({ heroId, maxTroops: 10000 })), skills: [] });

    for (let level = 1; level <= 3 && run.status === 'playing'; level++) {
      const myTeam = buildMyTeam(pool, { slots: strong.map((h, i) => ({ heroId: h, skillIds: [], position: POS[i] })) }, deps);
      const enemy = SLOTTED_HEROES.slice(90 + level * 3, 93 + level * 3).map((h, i) =>
        deps.buildGeneral(h.id, [], POS[i])
      );
      const res = settleLevel(run, pool, myTeam, enemy, level);
      run = res.run;
      pool = res.pool;
      saveRun(storage, { run, pool });
      expect(loadRun(storage)).toEqual({ run, pool });   // 每关后都不丢状态
    }
    expect(['cleared', 'playing', 'dead']).toContain(run.status);
    expect(run.level).toBeGreaterThanOrEqual(1);
  });

  it('越界卡一旦入池，下一次出战就在战报里留下贡献', () => {
    const heroes = SLOTTED_HEROES.slice(0, 3).map((h) => h.id);
    let pool = addCards(createPool({ heroes: heroes.map((heroId) => ({ heroId, maxTroops: 10000 })) }), ['rb_echo']);
    const myTeam = buildMyTeam(pool, { slots: heroes.map((h, i) => ({ heroId: h, skillIds: [], position: POS[i] })) }, deps);
    // rb_echo 需要队友释放主动战法才可能再发动 → 给每人配一个主动战法
    const withSkills = myTeam.map((g) => ({ ...g, activeSkillIds: [...g.activeSkillIds, 'jifeng_ershi'] }));
    const enemy = SLOTTED_HEROES.slice(200, 203).map((h, i) => deps.buildGeneral(h.id, [], POS[i]));
    const res = settleLevel(createRun(7, 3), pool, withSkills, enemy, 1);
    const rows = rogueContributions(res.report);
    expect(rows.some((r) => r.card === 'rb_echo')).toBe(true);
  });

  it('奖励池在 3 关内能产出越界卡（爽点可达）', () => {
    const pool = createPool({ heroes: [{ heroId: 'h479', maxTroops: 9000 }] });
    const catalogue = { heroes: [], skills: [], cards: ROGUE_CARDS };
    const seen = new Set<string>();
    for (let seed = 0; seed < 20; seed++) {
      for (const o of rollRewards(new Rng(seed), pool, 2, catalogue)) if (o.kind === 'card') seen.add(o.payload);
    }
    expect(seen.size).toBe(ROGUE_CARDS.length);
  });
});
```
